CREATE VIEW VW_STAGING_HR
AS
SELECT
    e.EMPLOYEE_ID                                AS EmployeeID,
    e.FIRST_NAME                                 AS FirstName,
    e.LAST_NAME                                  AS LastName,
    e.FIRST_NAME || ' ' || e.LAST_NAME           AS EmployeeName,
    e.EMAIL                                      AS Email,
    e.PHONE_NUMBER                               AS PhoneNumber,
    e.HIRE_DATE                                  AS HireDate,
    e.JOB_ID                                     AS JobID,
    j.JOB_TITLE                                  AS JobTitle,
    j.MIN_SALARY                                 AS MinSalary,
    j.MAX_SALARY                                 AS MaxSalary,
    e.SALARY                                     AS Salary,
    NVL(e.COMMISSION_PCT, 0)                     AS CommissionPct,
    (e.SALARY * 12) + NVL(e.COMMISSION_PCT,0)*e.SALARY AS AnnualComp,
    e.MANAGER_ID                                 AS ManagerID,
    m.FIRST_NAME || ' ' || m.LAST_NAME           AS ManagerName,
    e.DEPARTMENT_ID                              AS DepartmentID,
    d.DEPARTMENT_NAME                            AS DepartmentName,
    d.LOCATION_ID                                AS LocationID,
    l.CITY                                       AS City,
    l.STATE_PROVINCE                             AS StateProvince,
    c.COUNTRY_ID                                 AS CountryID,
    c.COUNTRY_NAME                               AS CountryName,
    r.REGION_ID                                  AS RegionID,
    r.REGION_NAME                                AS RegionName,
    -- Métricas derivadas para análisis
    TRUNC(MONTHS_BETWEEN(SYSDATE, e.HIRE_DATE)/12) AS TenureYears,
    jh.START_DATE                                AS JobStartDate,
    jh.END_DATE                                  AS JobEndDate,
    jh.DEPARTMENT_ID                             AS JobHistoryDeptID,
    jh.JOB_ID                                    AS JobHistoryJobID
FROM HR.EMPLOYEES e
LEFT JOIN HR.JOBS            j  ON e.JOB_ID         = j.JOB_ID
LEFT JOIN HR.EMPLOYEES       m  ON e.MANAGER_ID     = m.EMPLOYEE_ID
LEFT JOIN HR.DEPARTMENTS     d  ON e.DEPARTMENT_ID  = d.DEPARTMENT_ID
LEFT JOIN HR.LOCATIONS       l  ON d.LOCATION_ID    = l.LOCATION_ID
LEFT JOIN HR.COUNTRIES       c  ON l.COUNTRY_ID     = c.COUNTRY_ID
LEFT JOIN HR.REGIONS         r  ON c.REGION_ID      = r.REGION_ID
LEFT JOIN HR.JOB_HISTORY     jh ON e.EMPLOYEE_ID    = jh.EMPLOYEE_ID
ORDER BY e.EMPLOYEE_ID
