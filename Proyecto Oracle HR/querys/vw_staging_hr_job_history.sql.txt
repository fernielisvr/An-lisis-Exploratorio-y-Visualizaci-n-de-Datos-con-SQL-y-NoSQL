CREATE VIEW VW_STAGING_HR_JOB_HISTORY
AS
SELECT
    jh.EMPLOYEE_ID,
    e.FIRST_NAME || ' ' || e.LAST_NAME   AS EmployeeName,
    jh.START_DATE,
    jh.END_DATE,
    jh.JOB_ID,
    j.JOB_TITLE,
    jh.DEPARTMENT_ID,
    d.DEPARTMENT_NAME
FROM HR.JOB_HISTORY jh
JOIN HR.EMPLOYEES   e ON jh.EMPLOYEE_ID = e.EMPLOYEE_ID
JOIN HR.JOBS        j ON jh.JOB_ID      = j.JOB_ID
JOIN HR.DEPARTMENTS d ON jh.DEPARTMENT_ID = d.DEPARTMENT_ID
ORDER BY jh.EMPLOYEE_ID, jh.START_DATE
